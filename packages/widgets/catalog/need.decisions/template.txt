                            key={"priority-" + decision.id}
                            name={"choice-" + decision.id}
                            bind={"selections." + decision.id}
                            options={decision.options.map((option) => ({
                              label: option.label,
                              value: option.value,
                            }))}
                            direction={
                              sizeMode === "large" && index > 0 ? "row" : "col"
                            }
                            ariaLabel={decision.question}
                          />
                        </Show.ElseIf>
                        <Show.Else>
                          <Row gap={3} align="start">
                            <Show when={sizeMode === "large"}>
                              <Text
                                value={String(index + 1).padStart(2, "0")}
                                size="xl"
                                color="tertiary"
                                tabularNums
                              />
                            </Show>
                            <Col flex={1} minWidth={0} gap={2}>
                              <Show when={sizeMode === "medium"}>
                                <Caption value={labels.editorialEyebrow} />
                              </Show>
                              <Title
                                value={decision.question}
                                size={
                                  sizeMode === "medium"
                                    ? "xl"
                                    : index === 0
                                      ? "lg"
                                      : "md"
                                }
                              />
                              <Text
                                value={decision.context}
                                size="sm"
                                color="secondary"
                              />
                            </Col>
                          </Row>
                          <Col gap={1}>
                            <Each
                              of={decision.options}
                              item="option"
                              index="optionIndex"
                            >
                              <Pressable
                                key={option.value}
                                padding={{ x: 2, y: 2 }}
                                radius="sm"
                                background={
                                  selections[decision.id] === option.value
                                    ? "surface-secondary"
                                    : "transparent"
                                }
                                $onClickAction='{ patchState: set("selections." + decision.id, option.value) }'
                              >
                                <Row gap={3} align="start">
                                  <Text
                                    value={choiceLetters[optionIndex]}
                                    size="sm"
                                    color="secondary"
                                    weight="medium"
                                  />
                                  <Col flex={1} minWidth={0} gap={1}>
                                    <Text
                                      value={option.label}
                                      size="sm"
                                      weight="semibold"
                                    />
                                    <Show
                                      when={
                                        sizeMode === "medium" || index === 0
                                      }
                                    >
                                      <Text
                                        value={option.consequence}
                                        size="sm"
                                        color="secondary"
                                      />
                                    </Show>
                                  </Col>
                                  <Icon
                                    name={
                                      selections[decision.id] === option.value
                                        ? "check"
                                        : "arrow-up-right"
                                    }
                                    size="sm"
                                    color={
                                      selections[decision.id] === option.value
                                        ? "info"
                                        : "tertiary"
                                    }
                                  />
                                </Row>
                              </Pressable>
                            </Each>
                          </Col>
                        </Show.Else>
                      </Show>
                      <Show
                        when={
                          has(selectedOption) &&
                          direction !== "comparison" &&
                          !(
                            direction === "editorial" &&
                            (sizeMode === "medium" || index === 0)
                          )
                        }
                      >
                        <Text
                          key={"consequence-" + decision.id}
                          value={selectedOption.consequence}
                          size="sm"
                          color="secondary"
                        />
                      </Show>
                      <Col key={"metadata-" + decision.id} gap={1}>
                        <Show when={direction !== "inbox"}>
                          <Caption
                            value={
                              direction === "priority"
                                ? decision.agent
                                : decision.agent +
                                  " · " +
                                  labels.waiting.toLowerCase() +
                                  " " +
                                  decision.wait
                            }
                          />
                        </Show>
                        <Show
                          when={
                            direction !== "priority" &&
                            (sizeMode === "large" || decision.urgent)
                          }
                        >
                          <Row gap={1}>
                            <Icon
                              name="clock"
                              size="xs"
                              color={decision.urgent ? "warning" : "secondary"}
                            />
                            <Caption
                              value={decision.deadline}
                              color={decision.urgent ? "warning" : "secondary"}
                            />
                          </Row>
                        </Show>
                      </Col>
                      <Row key={"actions-" + decision.id} gap={2}>
                        <Button
                          label={labels.answer}
                          size="2xl"
                          color="info"
                          disabled={!has(selectedOption)}
                          $onClickAction='{ patchState: set("outcomes." + decision.id, "answered"), updateState: { feedback: decision.question + " — " + (selectedOption?.label || "") + ". Recorded only in this preview." } }'
                        />
                        <Button
                          label={labels.dismiss}
                          size="2xl"
                          color="secondary"
                          variant="ghost"
                          $onClickAction='{ patchState: set("outcomes." + decision.id, "dismissed"), updateState: { feedback: decision.question + " — dismissed from this preview." } }'
                        />
                      </Row>
                    </Col>
                  </Row>
                </Col>
              </Scope>
            </Each>
            <Show.Else>
              <EmptyState
                icon="check-circle"
                title={labels.emptyTitle}
                description={labels.emptyDescription}
                action={{ label: labels.reset, action: resetAction }}
              />
            </Show.Else>
          </Show>

        </Card>
      </Show>
    </Show>
  </Scope>
</Scope>
