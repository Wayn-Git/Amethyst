                              cycle.number === sessionNumber
                                ? "semibold"
                                : "normal"
                            }
                            color={
                              cycle.number === sessionNumber
                                ? "primary"
                                : "secondary"
                            }
                            tabularNums
                          />
                        </Row>
                        <Box
                          height={direction === "editorial" ? 2 : 4}
                          radius={direction === "editorial" ? "none" : "full"}
                          background={
                            cycle.number < sessionNumber ||
                            (cycle.number === sessionNumber &&
                              (phase === "break" || seconds === 0))
                              ? accent
                              : cycle.number === sessionNumber
                                ? "primary"
                                : "surface-tertiary"
                          }
                        />
                        <Caption value={cycle.shortSchedule} size="sm" />
                      </Col>
                    </Each>
                  </Row>
                  <Caption value={labels.timelineLegend} size="sm" />
                </Col>
              </Show>
              <Show when={widgetSize !== "small"}>
                <Row gap={2}>
                  <Button
                    label={seconds < duration ? labels.resume : labels.start}
                    iconStart="play"
                    size="2xl"
                    color="primary"
                    disabled={running || seconds === 0}
                    onClickAction={{
                      updateState: {
                        running: true,
                        deadline: now() + seconds * 1000,
                      },
                    }}
                  />
                  <Button
                    iconStart="pause"
                    ariaLabel={labels.pause}
                    uniform
                    size="2xl"
                    variant="soft"
                    disabled={!running}
                    onClickAction={{
                      updateState: {
                        running: false,
                        seconds: max(0, ceil((deadline - now()) / 1000)),
                      },
                    }}
                  />
                  <Button
                    iconStart="reload"
                    ariaLabel={labels.reset}
                    uniform
                    size="2xl"
                    variant="ghost"
                    disabled={seconds === 0}
                    onClickAction={{
                      updateState: {
                        running: false,
                        seconds: duration,
                        deadline: 0,
                      },
                    }}
                  />
                  <Spacer />
                  <Icon
                    name={seconds === 0 ? "check-circle" : "empty-circle"}
                    size="sm"
                    color={seconds === 0 ? accent : "tertiary"}
                  />
                </Row>
                <Show when={widgetSize === "medium"}>
                  <Caption
                    value={
                      seconds === 0
                        ? labels.complete
                        : String(completedTotal) +
                          labels.completedSuffix +
                          " · " +
                          modeLabel
                    }
                  />
                </Show>
              </Show>
              <Show when={widgetSize === "large" || seconds === 0}>
                <Box
                  padding={widgetSize === "small" ? 2 : 3}
                  background="surface-secondary"
                  radius={direction === "digital" ? "sm" : "md"}
                  gap={2}
                >
                  <Col gap={0}>
                    <Caption value={labels.upNext} />
                    <Text
                      value={
                        String(nextDuration / 60) +
                        labels.minuteUnit +
                        " " +
                        (phase === "focus"
                          ? labels.break.toLowerCase()
                          : labels.focus.toLowerCase())
                      }
                      size="sm"
                      weight="medium"
                    />
                  </Col>
                  <Show when={seconds === 0}>
                    <Button
                      label={
                        phase === "focus"
                          ? labels.beginBreak
                          : labels.beginFocus
                      }
                      size="2xl"
                      block
                      onClickAction={{
                        updateState: {
                          phase: phase === "focus" ? "break" : "focus",
                          seconds: nextDuration,
                          running: false,
                          deadline: 0,
                          completed: completed + (phase === "focus" ? 1 : 0),
                          dailyFocusBefore:
                            dailyFocusBefore +
                            (phase === "focus" ? duration : 0),
                          sessionNumber:
                            phase === "break"
                              ? sessionNumber === 4
                                ? 1
                                : sessionNumber + 1
                              : sessionNumber,
                        },
                      }}
                    />
                  </Show>
                </Box>
              </Show>
            </Box>
          </Card>
        </Row>
      </Scope>
    </Scope>
</Show>
